REPORT YRB_NEWSYNTAX_COND_02.

TABLES : VBRK.
*TYPES : BEGIN OF LTY_DATA,
*        VBELN TYPE VBELN_VF,
*        NETWR TYPE NETWR,
*        END OF LTY_DATA.
*
*DATA : LT_DATA TYPE TABLE OF LTY_DATA.
*DATA : LWA_DATA TYPE LTY_DATA.
*DATA : LV_TEXT TYPE CHAR20.
*SELECT-OPTIONS : S_VBELN FOR VBRK-VBELN.
*
*SELECT VBELN
*       NETWR
*       FROM   VBRK
*       INTO TABLE LT_data
*       WHERE VBELN IN S_VBELN.

*LOOP AT LT_DATA INTO LWA_DATA.
*  IF LWA_DATA-NETWR GE 0 AND LWA_DATA-NETWR LE 5000.
*     LV_TEXT = 'LOW PRIORITY'.
*  ELSEIF LWA_DATA-NETWR > 5000 AND LWA_DATA-NETWR < 10000.
*     LV_TEXT = 'MEDIUM PRIORITY'.
*  ELSE.
*    LV_TEXT = 'HIGH PRIORITY'.
*  ENDIF.
*    WRITE : / LWA_DATA-VBELN , LV_TEXT.
*ENDLOOP.


TYPES : BEGIN OF LTY_DATA,
        VBELN TYPE VBELN_VF,
        NETWR TYPE NETWR,
        END OF LTY_DATA.

DATA : LT_DATA TYPE TABLE OF LTY_DATA.
DATA : LWA_DATA TYPE LTY_DATA.
SELECT-OPTIONS : S_VBELN FOR VBRK-VBELN.

SELECT VBELN
       NETWR
       FROM   VBRK
       INTO TABLE LT_data
       WHERE VBELN IN S_VBELN.

LOOP AT LT_DATA INTO LWA_DATA.
 DATA(LV_TEXT) =    COND CHAR20( WHEN LWA_DATA-NETWR GE 0 AND LWA_DATA-NETWR LE 5000   THEN 'LOW PRIORITY'
                         WHEN LWA_DATA-NETWR > 5000 AND LWA_DATA-NETWR < 10000 THEN 'MEDIUM PRIORITY'
                         ELSE 'HIGH PRIORITY' ).
  WRITE : / LWA_DATA-VBELN , LV_TEXT.
ENDLOOP.
